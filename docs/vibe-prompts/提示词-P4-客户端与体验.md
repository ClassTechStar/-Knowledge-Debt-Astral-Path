### 3.7 公式 JS 实现（照抄，不要自己写）

```js
const sig=x=>1/(1+Math.exp(-x));
const clamp01=v=>Math.max(0,Math.min(1,v));
function empirical(n,s){return (s+0.5)/(n+1);}
function stability(streak,att){return K.stab0*(1+K.a*Math.max(streak,0))*(1+K.b*Math.log(1+Math.max(att,1)));}
function retention(age,streak,att){return Math.exp(-Math.max(age,0)/stability(streak,att));}
function prereqSup(arr){if(!arr||!arr.length)return .5;
  const mean=arr.reduce((a,b)=>a+b,0)/arr.length,mn=Math.min.apply(null,arr);
  return (1-K.mix)*mean+K.mix*mn;}
function prereqCeil(arr){return (!arr||!arr.length)?1:K.floor+(1-K.floor)*Math.min.apply(null,arr);}
function scoreV2(att,ok,conf,age,streak,prereq){
  const know=.85*empirical(att,ok)+.15*clamp01(conf/5);
  const raw=K.W.e*know+K.W.r*retention(age,streak,att)+K.W.p*prereqSup(prereq);
  return clamp01(Math.min(raw,prereqCeil(prereq)));
}
function impactV2(sf,st,w,tg,freq,down){
  if(!(freq>0))return 0;
  const ww=Number.isFinite(w)?Math.max(w,0):0;
  return K.impactBase*sig((sf-K.pg)*K.steep)*sig((K.cg-st)*K.steep)*ww*(tg?K.crossTg:1)
       *(1+K.cas*Math.log(1+Math.max(down,0)))*(1+K.vol*Math.log(1+Math.max(freq,0)));
}
function saleStep(s,acc,conf){
  const st0=s||{streak:0,att:0};
  const meets=(0.7*acc+0.3*clamp01(conf/5))>=K.saleBar&&acc>=K.saleAcc&&conf>=K.saleConf;
  const streak=meets?st0.streak+1:0, att=st0.att+1;
  const status=(streak>=K.saleStreak&&att>=K.saleMinAtt)?"cleared":(streak>0||att>0?"repairing":"open");
  return {status,streak,att,meets};
}
```

```kotlin
class MainActivity : AppCompatActivity() {
    private lateinit var web: WebView
    private lateinit var assetLoader: WebViewAssetLoader

    @SuppressLint("SetJavaScriptEnabled")
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        web = WebView(this).apply {
            setLayerType(android.view.View.LAYER_TYPE_HARDWARE, null)   // 性能：滚动更顺
        }
        setContentView(web)
        assetLoader = WebViewAssetLoader.Builder()
            .addPathHandler("/assets/", WebViewAssetLoader.AssetsPathHandler(this))
            .build()
        web.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true
            databaseEnabled = true
            allowFileAccess = true
            allowContentAccess = false
            useWideViewPort = true
            loadWithOverviewMode = true
            setSupportZoom(false); builtInZoomControls = false; displayZoomControls = false
            mixedContentMode = WebSettings.MIXED_CONTENT_ALWAYS_ALLOW
            cacheMode = WebSettings.LOAD_DEFAULT
            textZoom = 100                       // 防字体抖动
            mediaPlaybackRequiresUserGesture = false
            setGeolocationEnabled(false)
        }
        // WebChromeClient：拦截 alert 转 Toast（防白屏弹窗）
        // WebViewClient：shouldInterceptRequest 里把 appassets.androidplatform.net 交给 assetLoader
        val htmlRaw = assets.open("www/index.html").bufferedReader().use { it.readText() }
        val html = injectOffline(htmlRaw)
        web.loadDataWithBaseURL("https://appassets.androidplatform.net/assets/www/",
                                html, "text/html", "utf-8", null)
    }

    private fun injectOffline(raw: String): String {
        if (raw.contains("__ASTRALPATH_OFFLINE__=true")) return raw
        val flag = "<script>window.__ASTRALPATH_OFFLINE__=true;window.__ASTRALPATH_FORCE_ONLINE__=false;</script>"
        val i = raw.indexOf("<script>", ignoreCase = true)
        return if (i >= 0) raw.substring(0, i) + flag + raw.substring(i) else flag + raw
    }

    override fun onPause()  { web.onPause();  super.onPause() }
    override fun onResume() { super.onResume(); web.onResume() }
    override fun onDestroy(){ web.destroy();  super.onDestroy() }
}
```

## 10. 常见坑

| 坑 | 表现 | 解决 |
|---|---|---|
| **AAPT2265** | 安卓打包报资源名非法 | 项目路径/文件名**必须纯 ASCII**，不要有中文和空格 |
| 安卓闪退 | 一启动就崩 | 主题必须是 `Theme.AppCompat.Light.NoActionBar`；别用 Material Components 主题 |
| 中文豆腐块 | 手机上是方框 | 字体栈加 `sans-serif` 兜底 |
| 三端不同步 | 手机上和电脑上长得不一样 | **每次改完主 HTML 都跑 md5 比对**，写进发布脚本 |
| localStorage 爆掉 | 上传大教材后保存失败 | `save()` 里超限自动截断 `sourceText`(2000) / `body`(800) |
| 常量抄错 | 手机上 impact 和电脑不一致 | 24 项逐项比对，抄完找 P2 签字 |
| badge 出现黄色 | 违反"全绿"要求 | `partial` 也改成 `--ok-bg` |
| WebView 空白 | 安卓加载不了 assets | 用 `WebViewAssetLoader` + `appassets.androidplatform.net` 基址，不要 `file:///android_asset` |
| Inno Setup 找不到中文语言 | 编译报语言缺失 | 用 `compiler:Default.isl` 的 `chinesesimplified` |
